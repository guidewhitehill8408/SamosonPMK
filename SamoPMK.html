เงื่อนไขเวลาในการสั่งอาหาร

⚠️ ผู้เข้าพักต้องสั่งอาหารล่วงหน้าอย่างน้อย 4 วัน ก่อนวันเข้าพัก เพื่อให้เจ้าหน้าที่บ้านพักรับรองตระเตรียมวัตถุดิบและจัดทำอาหารสดใหม่ล่วงหน้า

เงื่อนไขและข้อปฏิบัติตามระเบียบ

  1. ระยะเวลาสั่งซื้อ: ต้องสั่งอาหารล่วงหน้าก่อนวันเข้าพักอย่างน้อย 4 วัน
  2. การเลือกห้องพัก: โปรดระบุหมายเลขห้องพัก (VIP 1 ถึง VIP 7) ในหน้าถัดไป
  3. ตัวเลือกประเภทอาหาร: สามารถเลือกประเภทเนื้อสัตว์ (หมู / ไก่ / ทะเล) หรือขนาดได้ในหน้าเมนูอาหาร
  4. ข้อมูลผู้สั่ง: ผู้เข้าพักต้องระบุชื่อ-นามสกุล, เบอร์โทรศัพท์ และห้องพักในขั้นตอนการสรุปรายการ
  5. การส่งข้อมูล: เมื่อกดรับรอง รายการจะถูกส่งไปยัง ไลน์ส่วนตัวของเจ้าหน้าที่บ้านพักรับรอง โดยอัตโนมัติ

Google Apps Script Integration

โค้ดสำหรับนำไปวางใน Google Apps Script เพื่อให้บันทึกจองลง Google Calendar และเด้ง LINE Notify หาเจ้าหน้าที่บ้านพักรับรองโดยอัตโนมัติ: 

function doPost(e) {
  var data = JSON.parse(e.postData.contents);
  var token = "YOUR_LINE_NOTIFY_TOKEN";
  
  // 1. Save to Google Calendar
  var cal = CalendarApp.getDefaultCalendar();
  cal.createEvent("จองบ้านพัก " + data.rooms.join(", "),
    new Date(data.checkin),
    new Date(data.checkout),
    { description: "หัวหน้าคณะ: " + data.headName + "\nโทร: " + data.phone }
  );

  // 2. Send LINE Notify to Officer
  var msg = "\n📌 มีการจองบ้านพักรับรองภูหมันขาว\nห้อง: " + data.rooms.join(", ") +
            "\nผู้เข้าพัก: " + data.headName + " (" + data.phone + ")";
            
  UrlFetchApp.fetch("https://notify-api.line.me/api/notify", {
    "method": "post",
    "headers": {"Authorization": "Bearer " + token},
    "payload": {"message": msg}
  });
}
